Create `src/app/features/fees/presentation/controllers/feesController.svelte.ts`.

This is a `.svelte.ts` file (Svelte 5 runes context). Follow the `createDisputeListController` pattern precisely.

The controller manages three independent concerns: global rules list, merchant rules list, and simulator. The `sheetOpen`, `editingRule`, `deletingId`, `confirmDeleteOpen` state is also managed here (consumed by Plan 02's FeeRuleSheet and ConfirmDialog).

```typescript
import { FeeService } from '$appmod/features/fees/services/FeeService';
import { FeeRepository } from '$appmod/features/fees/data/repositories/FeeRepository';
import type { FeeRule, FeeType, SimulateFeeResult, CreateFeeRulePayload, UpdateFeeRulePayload } from '$appmod/features/fees/domain/entities/FeeRule';
import { toast } from 'svelte-sonner';

interface FeesState {
  // Tab Global
  globalRules: FeeRule[];
  globalTotal: number;
  globalPage: number;
  globalPageSize: number;
  globalLoading: boolean;
  globalError: string | null;

  // Tab Por Merchant
  merchantId: string;
  merchantRules: FeeRule[];
  merchantLoading: boolean;
  merchantError: string | null;

  // Sheet (create/edit) — consumed by Plan 02
  sheetOpen: boolean;
  editingRule: FeeRule | null;
  saving: boolean;
  saveError: string | null;

  // Delete confirm — consumed by Plan 02
  deletingId: string | null;
  confirmDeleteOpen: boolean;
  deleting: boolean;

  // Simulator
  simFeeType: FeeType | '';
  simAmount: string;
  simMerchantId: string;
  simResult: SimulateFeeResult | null;
  simLoading: boolean;
  simError: string | null;
}

export function createFeesController() {
  const service = new FeeService(new FeeRepository());

  let state = $state<FeesState>({
    globalRules: [],
    globalTotal: 0,
    globalPage: 1,
    globalPageSize: 20,
    globalLoading: true,
    globalError: null,

    merchantId: '',
    merchantRules: [],
    merchantLoading: false,
    merchantError: null,

    sheetOpen: false,
    editingRule: null,
    saving: false,
    saveError: null,

    deletingId: null,
    confirmDeleteOpen: false,
    deleting: false,

    simFeeType: '',
    simAmount: '',
    simMerchantId: '',
    simResult: null,
    simLoading: false,
    simError: null,
  });

  // ── Global Rules ─────────────────────────────────────
  async function loadGlobalRules() {
    state.globalLoading = true;
    state.globalError = null;
    const result = await service.listRules(state.globalPage, state.globalPageSize);
    if (result.ok) {
      state.globalRules = result.value.items;
      state.globalTotal = result.value.total;
    } else {
      state.globalError = 'Não foi possível carregar as regras. Verifique sua conexão e tente novamente.';
    }
    state.globalLoading = false;
  }

  function setGlobalPage(p: number) {
    state.globalPage = p;
    loadGlobalRules();
  }

  // ── Merchant Rules ────────────────────────────────────
  async function loadMerchantRules(merchantId: string) {
    if (!merchantId) return;
    state.merchantLoading = true;
    state.merchantError = null;
    const result = await service.listMerchantRules(merchantId);
    if (result.ok) {
      state.merchantRules = result.value;
    } else {
      state.merchantError = 'Não foi possível carregar as regras. Verifique sua conexão e tente novamente.';
    }
    state.merchantLoading = false;
  }

  function setMerchantId(id: string) {
    state.merchantId = id;
    state.merchantRules = [];
    if (id) loadMerchantRules(id);
  }

  // ── Sheet ─────────────────────────────────────────────
  function openCreate(merchantId?: string) {
    state.editingRule = null;
    state.saveError = null;
    // Pass merchantId context so Sheet can pre-set merchant scope
    state.sheetOpen = true;
  }

  function openEdit(rule: FeeRule) {
    state.editingRule = rule;
    state.saveError = null;
    state.sheetOpen = true;
  }

  function closeSheet() {
    state.sheetOpen = false;
    state.editingRule = null;
    state.saveError = null;
  }

  async function saveRule(payload: CreateFeeRulePayload | UpdateFeeRulePayload, isEdit: boolean) {
    state.saving = true;
    state.saveError = null;
    let result;
    if (isEdit && state.editingRule) {
      result = await service.updateRule(state.editingRule.id, payload as UpdateFeeRulePayload);
    } else {
      result = await service.createRule(payload as CreateFeeRulePayload);
    }
    if (result.ok) {
      closeSheet();
      toast.success(isEdit ? 'Regra atualizada com sucesso.' : 'Regra criada com sucesso.');
      // Reload appropriate list
      if (state.merchantId) loadMerchantRules(state.merchantId);
      loadGlobalRules();
    } else {
      state.saveError = 'Erro ao salvar a regra. Verifique os dados e tente novamente.';
    }
    state.saving = false;
  }

  // ── Delete ────────────────────────────────────────────
  function openDeleteConfirm(id: string) {
    state.deletingId = id;
    state.confirmDeleteOpen = true;
  }

  function closeDeleteConfirm() {
    state.confirmDeleteOpen = false;
    state.deletingId = null;
  }

  async function deleteRule() {
    if (!state.deletingId) return;
    state.deleting = true;
    const result = await service.deleteRule(state.deletingId);
    if (result.ok) {
      closeDeleteConfirm();
      toast.success('Regra excluída.');
      if (state.merchantId) loadMerchantRules(state.merchantId);
      loadGlobalRules();
    } else {
      toast.error('Não foi possível excluir a regra. Tente novamente.');
      closeDeleteConfirm();
    }
    state.deleting = false;
  }

  // ── Simulator ─────────────────────────────────────────
  function setSimFeeType(v: FeeType | '') { state.simFeeType = v; }
  function setSimAmount(v: string) { state.simAmount = v; }
  function setSimMerchant(id: string) { state.simMerchantId = id; }

  async function runSimulate() {
    if (!state.simFeeType || !state.simAmount) return;
    state.simLoading = true;
    state.simError = null;
    state.simResult = null;
    const result = await service.simulate({
      feeType: state.simFeeType as FeeType,
      amount: Math.round(parseFloat(state.simAmount) * 100),  // R$ → centavos
      merchantId: state.simMerchantId || null,
    });
    if (result.ok) {
      state.simResult = result.value;
    } else {
      state.simError = 'Não foi possível calcular a simulação. Verifique os dados e tente novamente.';
    }
    state.simLoading = false;
  }

  return {
    get state() { return state; },
    loadGlobalRules,
    setGlobalPage,
    loadMerchantRules,
    setMerchantId,
    openCreate,
    openEdit,
    closeSheet,
    saveRule,
    openDeleteConfirm,
    closeDeleteConfirm,
    deleteRule,
    setSimFeeType,
    setSimAmount,
    setSimMerchant,
    runSimulate,
  };
}
```

<script lang="ts">
  import { onMount } from 'svelte';
  import * as Tabs from '$lib/components/ui/tabs/index.js';
  import { Button } from '$lib/components/ui/button';
  import DataTable from '$appmod/shared/widgets/DataTable.svelte';
  import SelectFilter from '$appmod/shared/widgets/filters/SelectFilter.svelte';
  import MerchantAutocomplete from '$appmod/features/transactions/shared/components/MerchantAutocomplete.svelte';
  import ConfirmDialog from '$appmod/shared/widgets/ConfirmDialog.svelte';
  import { createFeesController } from '../controllers/feesController.svelte';
  import { formatCurrency, formatBasisPoints } from '$appmod/shared/utils/formatters';
  import { hasPermission } from '$appmod/shared/guards/adminGuard';
  import { tokenStorage } from '$appmod/services/storage/tokenStorage';
  import type { AdminRole } from '$appmod/shared/guards/adminGuard';
  import type { FeeRule, FeeType } from '$appmod/features/fees/domain/entities/FeeRule';
  import { FEE_TYPE_LABELS, CALCULATION_LABELS } from '$appmod/features/fees/domain/entities/FeeRule';
  import { Pencil, Trash2 } from 'lucide-svelte';
  import type { ColumnDef } from '@tanstack/table-core';

  const role = tokenStorage.getAdminRole() as AdminRole | null;
  const canMutate = $derived(hasPermission(role, 'ADMIN'));

  const ctrl = createFeesController();

  let activeTab = $state('global');

  // Simulator fee type options
  const feeTypeOptions = Object.entries(FEE_TYPE_LABELS).map(([value, label]) => ({ value, label }));

  // DataTable columns — shared between global and merchant tabs
  const columns: ColumnDef<FeeRule>[] = [
    { id: 'feeType',        header: 'Tipo',        accessorKey: 'feeType' },
    { id: 'calculation',    header: 'Cálculo',      accessorKey: 'calculation' },
    { id: 'percentageRate', header: 'Percentual',   accessorKey: 'percentageRate' },
    { id: 'fixedAmount',    header: 'Valor Fixo',   accessorKey: 'fixedAmount' },
    { id: 'minMax',         header: 'Mín / Máx',    accessorKey: 'minFee' },
    { id: 'actions',        header: 'Ações',        accessorKey: 'id' },
  ];

  onMount(() => {
    ctrl.loadGlobalRules();
  });
</script>

      style="width: 100%; background: var(--color-surface-overlay); border: 1px solid var(--color-border); border-radius: var(--radius-md); padding: 8px 12px; color: var(--color-foreground); font-family: var(--font-mono); font-size: 0.875rem; box-sizing: border-box;"

    <MerchantAutocomplete
      value={ctrl.state.simMerchantId}
      onChange={(id) => ctrl.setSimMerchant(id)}
    />
  </div>

  <!-- Simular button -->
  <Button
    onclick={() => ctrl.runSimulate()}
    disabled={ctrl.state.simLoading || !ctrl.state.simFeeType || !ctrl.state.simAmount}
    style="width: 100%; background: var(--color-brand-magenta); border-radius: var(--radius-md); font-weight: 600; {ctrl.state.simLoading ? 'opacity: 0.7;' : ''}"
  >
    {ctrl.state.simLoading ? 'Calculando...' : 'Simular Taxa'}

  <!-- Result block -->
  {#if ctrl.state.simError}
