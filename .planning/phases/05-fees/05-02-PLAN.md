This component owns all form state internally. It receives `open`, `editingRule`, `saving`, `saveError`, `onClose`, and `onSave` as props.

**Props interface:**

```typescript
let {
  open = $bindable(false),
  editingRule = null,
  saving = false,
  saveError = null,
  merchantScope = null,        // pre-set merchantId when creating from "Por Merchant" tab
  onClose,
  onSave,
}: {
  open?: boolean;
  editingRule?: FeeRule | null;
  saving?: boolean;
  saveError?: string | null;
  merchantScope?: string | null;
  onClose: () => void;
  onSave: (payload: CreateFeeRulePayload | UpdateFeeRulePayload, isEdit: boolean) => void;
} = $props();
```

**Form state (all $state):**

```typescript
let feeType     = $state<FeeType | ''>('');
let calculation = $state<Calculation | ''>('');
let percentFloat = $state(0);    // user input in % — e.g. 2.5 means 2.5%
let reaisFloat   = $state(0);    // user input in R$ — e.g. 1.50
let minReais     = $state<number | null>(null);
let maxReais     = $state<number | null>(null);

// Derived hints (display only — NEVER sent to backend)
const basisPointsHint = $derived(Math.round(percentFloat * 100));
const centavosHint    = $derived(Math.round(reaisFloat * 100));

// Field visibility by calculation type
const showPercent = $derived(calculation === 'PERCENTAGE' || calculation === 'PERCENTAGE_PLUS_FIXED');
const showFixed   = $derived(calculation === 'FIXED' || calculation === 'PERCENTAGE_PLUS_FIXED');

// Sheet title
const isEdit = $derived(editingRule != null);
const sheetTitle = $derived(isEdit ? 'Editar Regra' : 'Nova Regra de Taxa');
const saveLabel  = $derived(isEdit ? 'Salvar Alterações' : 'Salvar Regra');

// Select trigger labels (Select.Value not exported — compute via $derived)
const feeTypeLabel = $derived(
  feeType ? (FEE_TYPE_LABELS[feeType as FeeType] ?? feeType) : 'Selecione o tipo...'
);
const calculationLabel = $derived(
  calculation ? (CALCULATION_LABELS[calculation as Calculation] ?? calculation) : 'Selecione o cálculo...'
);
```

**$effect for pre-fill and reset:**

```typescript
$effect(() => {
  if (open && editingRule) {
    feeType     = editingRule.feeType;
    calculation = editingRule.calculation;
    percentFloat = editingRule.percentageRate / 100;  // 250 → 2.50
    reaisFloat   = editingRule.fixedAmount   / 100;   // 350 → 3.50
    minReais = editingRule.minFee != null ? editingRule.minFee / 100 : null;
    maxReais = editingRule.maxFee != null ? editingRule.maxFee / 100 : null;
  } else if (open && !editingRule) {
    // Reset form for new rule
    feeType     = '';
    calculation = '';
    percentFloat = 0;
    reaisFloat   = 0;
    minReais     = null;
    maxReais     = null;
  }
});
```

**handleSubmit:**

```typescript
function handleSubmit() {
  if (!feeType || !calculation) return;
  const isEditMode = editingRule != null;
  const payload: CreateFeeRulePayload = {
    merchantId: editingRule?.merchantId ?? merchantScope ?? null,
    feeType: feeType as FeeType,
    calculation: calculation as Calculation,
    percentageRate: Math.round(percentFloat * 100),   // 2.5 → 250 basis points
    fixedAmount:    Math.round(reaisFloat   * 100),   // 1.5 → 150 centavos
    minFee:  minReais != null ? Math.round(minReais * 100) : null,
    maxFee:  maxReais != null ? Math.round(maxReais * 100) : null,
  };
  onSave(payload, isEditMode);
}
```
