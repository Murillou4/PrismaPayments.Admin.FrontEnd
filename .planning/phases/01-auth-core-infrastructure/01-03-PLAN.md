<script lang="ts">
  interface Props {
    open: boolean;
    title: string;
    description?: string;
    confirmLabel?: string;
    cancelLabel?: string;
    destructive?: boolean;
    requiresReason?: boolean;
    reasonLabel?: string;
    onconfirm: (reason?: string) => void;
    oncancel: () => void;
  }

  let {
    open,
    title,
    description = '',
    confirmLabel = 'Confirmar',
    cancelLabel = 'Cancelar',
    destructive = false,
    requiresReason = false,
    reasonLabel = 'MOTIVO',
    onconfirm,
    oncancel
  }: Props = $props();

  let dialog = $state<HTMLDialogElement | null>(null);
  let reason = $state('');

  // Controle programático via $effect — CSS transitions interrompíveis
  $effect(() => {
    if (!dialog) return;
    if (open) {
      dialog.showModal();
    } else {
      reason = '';  // limpar ao fechar
      if (dialog.open) dialog.close();
    }
  });

  function handleConfirm() {
    onconfirm(requiresReason ? reason : undefined);
  }

  function handleCancel() {
    oncancel();
  }

  // Fechar ao clicar no backdrop (fora do panel)
  function handleBackdropClick(e: MouseEvent) {
    if (e.target === dialog) {
      handleCancel();
    }
  }

  const canConfirm = $derived(!requiresReason || reason.trim().length > 0);
</script>
